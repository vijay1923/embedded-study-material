
//uart driver//
#include<lpc21xx.h>
#include<stdio.h>
#include"header.h"
#define THRE ((U0LSR>>5)&1)
#define RDR  (U0LSR&1)    //RECEIVING FLG


void uart0_init(unsigned int baud)
{
	//GETTING BAUDRATE BY USING PCLK//
	unsigned int result=0,pclk;
	int a[]={15,60,30,15,15};
	pclk=a[VPBDIV]*1000000;
	result=pclk/(16*baud);
	// SETTING BAUD RATE FOR COMMUNICATION//
	PINSEL0 |= 5;
	U0LCR=0X83;    //8N1 AND DLAB=1 that means baudrate setting unlock and Tx Rx data locked 
	U0DLM=(result>>8)&0xff;
	U0DLL=result&0x00ff;
	U0LCR=3;  // 	8N1 AND DLAB=0 that means baudrate setting locked and Tx Rx can be done.
}

// TRANSMIT 1BYTE OF DATA 
  // transmission flag 
void uart0_tx(unsigned char data)
{
	U0THR=data;
	while(THRE==0);
}

//RECIEVE 1BYTE OF DATA
unsigned char uart0_rx()
{
	while(RDR==0);
	return U0RBR;
}

//string print 
void uart0_tx_string(char *ptr)
{
	while(*ptr)
	{
		U0THR=*ptr;
		while(THRE==0);
    ptr++;
	}
}

//print int num
void uart0_tx_integer(int temp)
{
	unsigned char a[10]={0};
	sprintf(a,"%d",temp);
	uart0_tx_string(a);
}

//print float no.
void uart0_tx_float(float temp)
{
	char a[10]={0};
	sprintf(a,"%.1f",temp);
	uart0_tx_string(a);
}

//receive string from other controller
char * uart0_rx_string( char *s,int len)
{
	int i;
	for(i=0;i<len;i++)
	{
		s[i]=uart0_rx();
		if(s[i]=='\r');
		{
		  s[i]=0;
			break;
		}
	}
		return s;
}

void adc_init(void )
{
PINSEL0|=15400000;
ADCR=0X00200400;
}
unsigned int adc_read(unsigned char ch)
{
	unsigned int result;
ADCR|=1<<ch;
ADCR|=1<<24;
while(((ADDR>>31)&31)==0);
ADCR^=1<<24;
ADCR^=1<<ch;
result=(ADDR>>6)&0x3ff;
return result;
}

#include<lpc21xx.h>

void lcd_data(unsigned char data)
{
IOCLR1=0xFE<<16;
IOSET1=(data & 0xF0)<<16;
IOSET1=1<<17;
IOCLR1=1<<18;
IOSET1=1<<19;
delay_ms(2);
IOCLR1=1<<19;

IOCLR1=0xFE<<16;
IOSET1=(data & 0x0F)<<20;
IOSET1=1<<17;
IOCLR1=1<<18;
IOSET1=1<<19;
delay_ms(2);
IOCLR1=1<<19;
}
void lcd_cmd(unsigned char cmd)
{
IOCLR1=0xFE<<16;
IOSET1=(cmd& 0xF0)<<16;
IOCLR1=1<<17;    // RS=1
IOCLR1=1<<18;	 // RW=0
IOSET1=1<<19;	// EN=1
delay_ms(2);
IOCLR1=1<<19;

IOCLR1=0xFE<<16;
IOSET1=(cmd& 0x0F)<<20;
IOCLR1=1<<17;    // RS=1
IOCLR1=1<<18;	 // RW=0
IOSET1=1<<19;	// EN=1
delay_ms(2);
IOCLR1=1<<19;
}

void lcd_integer(long long int num)
{
int a[15],i;
if(num==0)
{
lcd_data('0');
return ;
}
if(num<0)
{
num=-num;
lcd_data('-');
}
i=0;
while(num>0)
{
a[i]=num%10+48;
num=num/10;
i++;
}
for(i=i-1;i>=0;i--)
{

}
}

void EXTINT0_HANDLER(void)__irq
{
   static int c;
   if(c%2==0)
   IOCLR0=1<<17;
   else
   IOSET0=1<<17;
   c++;
   EXTINT=1;
   VICVectAddr=0;
}

//blinking
void uart0_Handler(void)__irq    //// main fuction 
{
	 int r=U0IIR;        // store recive adress
	r=r&0x0E;            // addresss
	if(r==4) // recived data
	{
		op=U0RBR;  // char stored recived buffer
		if(op=='a')   // condi 1st
			flag1=1;
		else if(op=='b') // condn 2nd
			flag1=0;
		U0THR=op;        // store op in transmit buffer
	}
	VICVectAddr=0;
}
