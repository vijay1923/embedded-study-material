/*

1.create node 
2.scan the data
3.update newely created node next address
4.update headptr
*/
void add_bigin(**ptr)
{
  SLL *new;
new=malloc(sizeof(SLL)); 	// create new node 
printf("Enter rollno,name and marks\n");
scanf("%d%s%f",&new->rollno,&new->name,&new->marks);
new->next=*ptr;		// update newely created node next address 
*ptr=new;
}

// printing all node

{
 if(ptr==0)
 {
  printf("Records not found\n");
   return 0;
 }

while(ptr)
{
 printf("%d%s%f",ptr->rollno,ptr->name,ptr->mark);

}

// counting all nodes
int count_node(SLL *ptr)
{
 int c=0;
while(ptr)
{
 c++;
ptr=ptr->next;
}

return c;
}
